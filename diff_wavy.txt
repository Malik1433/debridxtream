+    override fun onKeyDown(keyCode: Int, event: KeyEvent?): Boolean {
+        if (duration <= 0) return super.onKeyDown(keyCode, event)
+
+        val increment = 15000L // 15 seconds
+        when (keyCode) {
+            KeyEvent.KEYCODE_DPAD_LEFT -> {
+                val newPos = (position - increment).coerceAtLeast(0L)
+                listeners.forEach { it.onScrubStart(this, position) }
+                listeners.forEach { it.onScrubStop(this, newPos, false) }
+                return true
+            }
+            KeyEvent.KEYCODE_DPAD_RIGHT -> {
+                val newPos = (position + increment).coerceAtMost(duration)
+                listeners.forEach { it.onScrubStart(this, position) }
+                listeners.forEach { it.onScrubStop(this, newPos, false) }
+                return true
+            }
+            KeyEvent.KEYCODE_ENTER, KeyEvent.KEYCODE_DPAD_CENTER -> {
+                // If the user clicks the seek bar, maybe toggle play/pause?
+                // For now, let's just consume it to avoid accidental navigation
+                return true
+            }
+        }
+        return super.onKeyDown(keyCode, event)
+    }
